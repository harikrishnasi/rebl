export async function callAI(prompt, systemPrompt = '') {
  const fullPrompt = systemPrompt ? systemPrompt + '\n\n' + prompt : prompt;
  const res = await fetch(
    `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${import.meta.env.VITE_GEMINI_API_KEY}`,
    { method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ contents:[{parts:[{text:fullPrompt}]}], generationConfig:{temperature:0.8, maxOutputTokens:800} }) }
  );
  const data = await res.json();
  if (data.error) throw new Error(data.error.message);
  return data.candidates[0].content.parts[0].text;
}
export async function callAIJSON(prompt, systemPrompt = '') {
  const text = await callAI(prompt + '\n\nReturn ONLY valid JSON. No markdown. No backticks.', systemPrompt);
  return JSON.parse(text.replace(/```json|```/g,'').trim());
}

=== SUPABASE SQL (DO NOT BUILD IN REACT - RUN THIS SEPARATELY IN SUPABASE SQL EDITOR) ===

create table public.profiles (id uuid references auth.users on delete cascade primary key, username text unique not null, display_name text, bio text, avatar_url text, city text, role text default 'collector' check (role in ('collector','brand','admin')), collector_score integer default 0, is_pro boolean default false, collector_archetype text, archetype_description text, signature_phrase text, created_at timestamptz default now());
create table public.items (id uuid default gen_random_uuid() primary key, owner_id uuid references public.profiles(id) on delete cascade, name text not null, brand text not null, edition text, serial_number text, purchase_date date, purchase_price numeric, image_url text, ai_story text, user_story text, is_verified boolean default false, is_for_sale boolean default false, category text default 'sneakers', vibe_tags text[] default '{}', estimated_value numeric, created_at timestamptz default now());
create table public.brands (id uuid default gen_random_uuid() primary key, name text not null, slug text unique not null, logo_url text, description text, tier text default 'starter', limited_edition_strategies jsonb default '[]', created_at timestamptz default now());
create table public.brand_categories (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id) on delete cascade, category text not null, is_primary boolean default false);
create table public.drops (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id), name text not null, description text, image_url text, total_units integer, price numeric, drop_date timestamptz, status text default 'upcoming', access_type text default 'public', extra_fields jsonb default '{}', created_at timestamptz default now());
create table public.purchases (id uuid default gen_random_uuid() primary key, user_id uuid references public.profiles(id), drop_id uuid references public.drops(id), item_id uuid references public.items(id), purchase_price numeric, story_completed boolean default false, community_joined boolean default false, created_at timestamptz default now());
create table public.product_stories (id uuid default gen_random_uuid() primary key, drop_id uuid references public.drops(id) on delete cascade, brand_id uuid references public.brands(id), headline text, origin_story text, design_intent text, limited_edition_reason text, behind_scenes text, published boolean default false);
create table public.customer_tiers (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id) on delete cascade, name text not null, tier_level integer not null, color text default 'CCCCCC', min_spend numeric default 0, min_purchases integer default 0, min_collector_score integer default 0, min_story_completions integer default 0, require_all boolean default false, perks jsonb default '[]', has_backstage_access boolean default false);
create table public.collector_tier_assignments (id uuid default gen_random_uuid() primary key, collector_id uuid references public.profiles(id), brand_id uuid references public.brands(id), tier_id uuid references public.customer_tiers(id), total_spend numeric default 0, purchase_count integer default 0, unique(collector_id, brand_id));
create table public.campaigns (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id), name text not null, type text, target_tier_ids uuid[], message_subject text, message_body text, status text default 'draft', sent_at timestamptz, open_count integer default 0);
create table public.post_purchase_contacts (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id), purchase_id uuid references public.purchases(id), collector_id uuid references public.profiles(id), trigger text, message text, sent_at timestamptz, opened boolean default false);
create table public.brand_subdomains (id uuid default gen_random_uuid() primary key, brand_id uuid references public.brands(id) unique, subdomain text unique not null, custom_domain text, theme_primary text default '0F0F1A', theme_accent text default 'E63946', hero_headline text, hero_subheadline text);
